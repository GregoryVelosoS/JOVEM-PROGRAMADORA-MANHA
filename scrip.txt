-- CRIAR UMA TABELA
CREATE TABLE tb_usuarios(
	id_usuarios INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(255) NOT NULL,
    dt_nasc DATE, 
    email VARCHAR(100) NOT NULL, 
    senha VARCHAR(45) NOT NULL,
    qtd_esposas BIGINT,
    esta_ativo TINYINT, 
    altura DECIMAL(5,2)
);
